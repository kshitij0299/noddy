## Helper scripts (copies; the originals sit in /home/user/.noddy-tools/ on the cloud VM)

`browser.mjs`:
```js
// Shared headless-browser helper for Noddy agents (not part of the repo).
// import { launch } from '/home/user/.noddy-tools/browser.mjs';
// const { browser, page, errors } = await launch({ width: 1440, height: 900 });
import { createRequire } from 'node:module';
const require = createRequire('/opt/node-tools/node_modules/');
export const { chromium } = require('playwright');
export async function launch({ width = 1440, height = 900 } = {}) {
	// Pass the proxy as a raw Chromium flag so loopback (the vite server) is NOT proxied
	// while Google Fonts still load through the agent proxy.
	const args = process.env.HTTPS_PROXY ? ['--proxy-server=' + process.env.HTTPS_PROXY] : [];
	const browser = await chromium.launch({ args });
	const page = await browser.newPage({ ignoreHTTPSErrors: true, viewport: { width, height } });
	const errors = [];
	page.on('console', (m) => m.type() === 'error' && errors.push(m.text()));
	page.on('pageerror', (e) => errors.push('pageerror: ' + e.message));
	page.on('requestfailed', (r) => {
		if (!/googletagmanager|google-analytics/.test(r.url())) errors.push('requestfailed: ' + r.url());
	});
	return { browser, page, errors };
}
```

`shot.mjs`:
```js
// usage: node /home/user/.noddy-tools/shot.mjs <url> <out.png> [width] [height]
import { launch } from './browser.mjs';
const [url, out, w = '1440', h = '900'] = process.argv.slice(2);
const { browser, page, errors } = await launch({ width: +w, height: +h });
await page.goto(url, { waitUntil: 'networkidle' });
await page.waitForTimeout(800);
await page.screenshot({ path: out });
console.log('errors:', errors.length, errors.slice(0, 5));
await browser.close();
```
